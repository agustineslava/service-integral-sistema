. Endpoint público en backend/server.js
Agregá en tu backend/server.js este endpoint sin verificarToken, para que cualquier celular que lea el QR pueda consultar los datos básicos del repuesto:

// Endpoint público de consulta rápida por QR (no requiere login)
app.get('/api/consulta-publica/:id', (req, res) => {
    const { id } = req.params;
    const sql = `
        SELECT id, nombre, precio, stock, categoria, distribuidor, 
               codigoDistribuidor, descripcion, imagen_1 
        FROM productos 
        WHERE id = ?
    `;
    db.query(sql, [id], (err, rows) => {
        if (err) {
            console.error('Error en consulta pública:', err);
            return res.status(500).json({ error: 'Error en el servidor' });
        }
        if (rows.length === 0) {
            return res.status(404).json({ error: 'Producto no encontrado' });
        }
        res.json(rows[0]);
    });
});

function abrirModalEtiqueta(id) {
    const prod = productosLocales.find(p => p.id === id);
    if (!prod) return;

    document.getElementById('etiqueta-nombre').textContent = prod.nombre;
    document.getElementById('etiqueta-codigo').textContent = `COD: ${prod.codigoDistribuidor || 'ART-' + prod.id}`;
    document.getElementById('etiqueta-distribuidor').textContent = `$${parseFloat(prod.precio).toFixed(2)}`;

    // URL que abrirá el celular al escanear
    const urlConsulta = `${window.location.origin}/consulta.html?id=${prod.id}`;

    // Renderizar código QR en el canvas
    const canvasQR = document.getElementById('codigo-barras-svg');
    if (canvasQR) {
        // Limpiamos y reemplazamos por un canvas QR
        canvasQR.outerHTML = '<canvas id="codigo-qr-canvas" style="width: 60px; height: 60px;"></canvas>';
        QRCode.toCanvas(document.getElementById('codigo-qr-canvas'), urlConsulta, {
            width: 70,
            margin: 1,
            color: { dark: "#000000", light: "#ffffff" }
        }, (err) => {
            if (err) console.error(err);
        });
    }

    document.getElementById('modal-etiqueta').style.display = 'block';
}
